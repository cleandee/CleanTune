# CleanTune - Warzone / BO7 - V1 Competitive (default)
# Pre-HeSuVi chain. Runs on the raw 7.1 game mix before virtual surround.
# Goal: lift footsteps, tame explosions and low rumble, keep headroom.
# Channels: L R C LFE RL RR SL SR

# Headroom so the boosts below never clip
Preamp: -7 dB

# ---- All channels ----
Channel: all
# Remove sub rumble headphones cannot reproduce cleanly
Filter: ON HP Fc 28 Hz Q 0.707
# Tame explosions, vehicles and killstreak bass
Filter: ON LSC Fc 120 Hz Gain -4 dB Q 0.7
# Clear the low-mid mud that masks movement
Filter: ON PK Fc 300 Hz Gain -2.5 dB Q 1.0
# Footstep body - boots on hard floors
Filter: ON PK Fc 1100 Hz Gain 1.5 dB Q 1.2
# Footstep presence and gear rustle - the main lift
Filter: ON PK Fc 3200 Hz Gain 4 dB Q 1.1
# Pull back harshness from gunfire crack
Filter: ON PK Fc 6500 Hz Gain -2 dB Q 2.0
# Slight top-end trim
Filter: ON HSC Fc 10000 Hz Gain -1 dB Q 0.7

# ---- Sub channel ----
# Halve the LFE channel so explosions do not swamp the mix
Channel: LFE
Preamp: -6 dB

# ---- Rear and side channels ----
# Small extra lift so enemies behind you stand out
Channel: RL RR SL SR
Filter: ON PK Fc 2800 Hz Gain 1.5 dB Q 1.0

Channel: all
